// Research probe, not a test of library code: how many multiply-accumulates per second plain WGSL
// neural layers reach on the test GPU. See dev-docs/research/new-contributors-2026-11-11.

import {getWebGPUTestDevice} from '@luma.gl/core/test-utils';
import {expect, it} from 'vitest';
import {requireTestDevice} from '../../src/testing/require-test-device';

const WORKGROUP_SIZE = 256;
const MAXIMUM_DISPATCH_X = 32768;

type ProbeCase = {
  label: string;
  multiplyAccumulateCount: number;
  invocationCount: number;
  inputValues: Float32Array;
  weightValues: Float32Array;
  outputLength: number;
  getSource: (dispatchX: number) => string;
  getReference: (outputIndex: number) => number;
};

function fillPseudoRandom(values: Float32Array, seed: number, scale: number): Float32Array {
  let state = seed >>> 0;
  for (let index = 0; index < values.length; index++) {
    state = (Math.imul(state, 1664525) + 1013904223) >>> 0;
    values[index] = (state / 4294967296 - 0.5) * scale;
  }
  return values;
}

function getDispatchPrelude(invocationCount: number, dispatchX: number): string {
  return /* wgsl */ `
const INVOCATION_COUNT = ${invocationCount}u;
const DISPATCH_X = ${dispatchX}u;
fn getInvocationIndex(workgroupId: vec3u, localIndex: u32) -> u32 {
  return (workgroupId.y * DISPATCH_X + workgroupId.x) * ${WORKGROUP_SIZE}u + localIndex;
}
@group(0) @binding(0) var<storage, read> inputValues: array<f32>;
@group(0) @binding(1) var<storage, read> weightValues: array<f32>;
@group(0) @binding(2) var<storage, read_write> outputValues: array<f32>;
`;
}

/** One invocation per output element: a dense layer with bias and ReLU. */
function createDenseElementCase(
  rowCount: number,
  inputSize: number,
  outputSize: number
): ProbeCase {
  const inputValues = fillPseudoRandom(new Float32Array(rowCount * inputSize), 1, 2);
  const weightValues = fillPseudoRandom(
    new Float32Array(outputSize * inputSize + outputSize),
    2,
    0.5
  );
  const invocationCount = rowCount * outputSize;
  return {
    label: `dense layer, one invocation per output: ${rowCount} rows, ${inputSize} -> ${outputSize}`,
    multiplyAccumulateCount: rowCount * inputSize * outputSize,
    invocationCount,
    inputValues,
    weightValues,
    outputLength: invocationCount,
    getSource: dispatchX => /* wgsl */ `
${getDispatchPrelude(invocationCount, dispatchX)}
const INPUT_SIZE = ${inputSize}u;
const OUTPUT_SIZE = ${outputSize}u;
@compute @workgroup_size(${WORKGROUP_SIZE})
fn main(@builtin(workgroup_id) workgroupId: vec3u, @builtin(local_invocation_index) localIndex: u32) {
  let index = getInvocationIndex(workgroupId, localIndex);
  if (index >= INVOCATION_COUNT) { return; }
  let row = index / OUTPUT_SIZE;
  let outputChannel = index % OUTPUT_SIZE;
  var sum = weightValues[OUTPUT_SIZE * INPUT_SIZE + outputChannel];
  for (var inputChannel = 0u; inputChannel < INPUT_SIZE; inputChannel++) {
    sum += weightValues[outputChannel * INPUT_SIZE + inputChannel] * inputValues[row * INPUT_SIZE + inputChannel];
  }
  outputValues[index] = max(sum, 0.0);
}`,
    getReference: outputIndex => {
      const row = Math.floor(outputIndex / outputSize);
      const outputChannel = outputIndex % outputSize;
      let sum = weightValues[outputSize * inputSize + outputChannel];
      for (let inputChannel = 0; inputChannel < inputSize; inputChannel++) {
        sum +=
          weightValues[outputChannel * inputSize + inputChannel] *
          inputValues[row * inputSize + inputChannel];
      }
      return Math.max(sum, 0);
    }
  };
}

/** One invocation per row: the row is copied to a local array, then every output is written. */
function createDenseRowCase(rowCount: number, inputSize: number, outputSize: number): ProbeCase {
  const base = createDenseElementCase(rowCount, inputSize, outputSize);
  return {
    ...base,
    label: `dense layer, one invocation per row: ${rowCount} rows, ${inputSize} -> ${outputSize}`,
    invocationCount: rowCount,
    getSource: dispatchX => /* wgsl */ `
${getDispatchPrelude(rowCount, dispatchX)}
const INPUT_SIZE = ${inputSize}u;
const OUTPUT_SIZE = ${outputSize}u;
@compute @workgroup_size(${WORKGROUP_SIZE})
fn main(@builtin(workgroup_id) workgroupId: vec3u, @builtin(local_invocation_index) localIndex: u32) {
  let row = getInvocationIndex(workgroupId, localIndex);
  if (row >= INVOCATION_COUNT) { return; }
  var rowValues: array<f32, ${inputSize}>;
  for (var inputChannel = 0u; inputChannel < INPUT_SIZE; inputChannel++) {
    rowValues[inputChannel] = inputValues[row * INPUT_SIZE + inputChannel];
  }
  for (var outputChannel = 0u; outputChannel < OUTPUT_SIZE; outputChannel++) {
    var sum = weightValues[OUTPUT_SIZE * INPUT_SIZE + outputChannel];
    for (var inputChannel = 0u; inputChannel < INPUT_SIZE; inputChannel++) {
      sum += weightValues[outputChannel * INPUT_SIZE + inputChannel] * rowValues[inputChannel];
    }
    outputValues[row * OUTPUT_SIZE + outputChannel] = max(sum, 0.0);
  }
}`
  };
}

/**
 * A whole coordinate network in one kernel, one invocation per point: two coordinates, a Fourier
 * encoding of `2 * 2 * frequencyCount` features, two hidden layers with ReLU and one output.
 */
function createCoordinateNetworkCase(
  pointCount: number,
  frequencyCount: number,
  hiddenSize: number
): ProbeCase {
  const featureCount = 4 * frequencyCount;
  const firstOffset = 0;
  const secondOffset = firstOffset + hiddenSize * featureCount + hiddenSize;
  const thirdOffset = secondOffset + hiddenSize * hiddenSize + hiddenSize;
  const weightCount = thirdOffset + hiddenSize + 1;
  const inputValues = fillPseudoRandom(new Float32Array(pointCount * 2), 3, 2);
  const weightValues = fillPseudoRandom(new Float32Array(weightCount), 4, 0.4);
  return {
    label: `coordinate network in one kernel: ${pointCount} points, ${featureCount} -> ${hiddenSize} -> ${hiddenSize} -> 1`,
    multiplyAccumulateCount:
      pointCount * (featureCount * hiddenSize + hiddenSize * hiddenSize + hiddenSize),
    invocationCount: pointCount,
    inputValues,
    weightValues,
    outputLength: pointCount,
    getSource: dispatchX => /* wgsl */ `
${getDispatchPrelude(pointCount, dispatchX)}
const FREQUENCY_COUNT = ${frequencyCount}u;
const FEATURE_COUNT = ${featureCount}u;
const HIDDEN_SIZE = ${hiddenSize}u;
const SECOND_OFFSET = ${secondOffset}u;
const THIRD_OFFSET = ${thirdOffset}u;
@compute @workgroup_size(${WORKGROUP_SIZE})
fn main(@builtin(workgroup_id) workgroupId: vec3u, @builtin(local_invocation_index) localIndex: u32) {
  let point = getInvocationIndex(workgroupId, localIndex);
  if (point >= INVOCATION_COUNT) { return; }
  let position = vec2f(inputValues[point * 2u], inputValues[point * 2u + 1u]);
  var features: array<f32, ${featureCount}>;
  var frequency = 3.14159265;
  for (var band = 0u; band < FREQUENCY_COUNT; band++) {
    features[band * 4u] = sin(position.x * frequency);
    features[band * 4u + 1u] = cos(position.x * frequency);
    features[band * 4u + 2u] = sin(position.y * frequency);
    features[band * 4u + 3u] = cos(position.y * frequency);
    frequency *= 2.0;
  }
  var first: array<f32, ${hiddenSize}>;
  for (var unit = 0u; unit < HIDDEN_SIZE; unit++) {
    var sum = weightValues[HIDDEN_SIZE * FEATURE_COUNT + unit];
    for (var feature = 0u; feature < FEATURE_COUNT; feature++) {
      sum += weightValues[unit * FEATURE_COUNT + feature] * features[feature];
    }
    first[unit] = max(sum, 0.0);
  }
  var second: array<f32, ${hiddenSize}>;
  for (var unit = 0u; unit < HIDDEN_SIZE; unit++) {
    var sum = weightValues[SECOND_OFFSET + HIDDEN_SIZE * HIDDEN_SIZE + unit];
    for (var source = 0u; source < HIDDEN_SIZE; source++) {
      sum += weightValues[SECOND_OFFSET + unit * HIDDEN_SIZE + source] * first[source];
    }
    second[unit] = max(sum, 0.0);
  }
  var result = weightValues[THIRD_OFFSET + HIDDEN_SIZE];
  for (var unit = 0u; unit < HIDDEN_SIZE; unit++) {
    result += weightValues[THIRD_OFFSET + unit] * second[unit];
  }
  outputValues[point] = result;
}`,
    getReference: point => {
      const features = new Float64Array(featureCount);
      let frequency = Math.fround(3.14159265);
      for (let band = 0; band < frequencyCount; band++) {
        features[band * 4] = Math.sin(inputValues[point * 2] * frequency);
        features[band * 4 + 1] = Math.cos(inputValues[point * 2] * frequency);
        features[band * 4 + 2] = Math.sin(inputValues[point * 2 + 1] * frequency);
        features[band * 4 + 3] = Math.cos(inputValues[point * 2 + 1] * frequency);
        frequency *= 2;
      }
      const first = new Float64Array(hiddenSize);
      for (let unit = 0; unit < hiddenSize; unit++) {
        let sum = weightValues[hiddenSize * featureCount + unit];
        for (let feature = 0; feature < featureCount; feature++) {
          sum += weightValues[unit * featureCount + feature] * features[feature];
        }
        first[unit] = Math.max(sum, 0);
      }
      const second = new Float64Array(hiddenSize);
      for (let unit = 0; unit < hiddenSize; unit++) {
        let sum = weightValues[secondOffset + hiddenSize * hiddenSize + unit];
        for (let source = 0; source < hiddenSize; source++) {
          sum += weightValues[secondOffset + unit * hiddenSize + source] * first[source];
        }
        second[unit] = Math.max(sum, 0);
      }
      let result = weightValues[thirdOffset + hiddenSize];
      for (let unit = 0; unit < hiddenSize; unit++) {
        result += weightValues[thirdOffset + unit] * second[unit];
      }
      return result;
    }
  };
}

/** A 3 by 3 convolution with zero padding, channel-major layout, one invocation per output value. */
function createConvolutionCase(size: number, inputChannels: number, outputChannels: number) {
  const cellCount = size * size;
  const inputValues = fillPseudoRandom(new Float32Array(cellCount * inputChannels), 5, 2);
  const weightValues = fillPseudoRandom(
    new Float32Array(outputChannels * inputChannels * 9 + outputChannels),
    6,
    0.2
  );
  const invocationCount = cellCount * outputChannels;
  const probeCase: ProbeCase = {
    label: `3x3 convolution: ${size} x ${size}, ${inputChannels} -> ${outputChannels} channels`,
    multiplyAccumulateCount: cellCount * 9 * inputChannels * outputChannels,
    invocationCount,
    inputValues,
    weightValues,
    outputLength: invocationCount,
    getSource: dispatchX => /* wgsl */ `
${getDispatchPrelude(invocationCount, dispatchX)}
const SIZE = ${size}u;
const CELL_COUNT = ${cellCount}u;
const INPUT_CHANNELS = ${inputChannels}u;
const OUTPUT_CHANNELS = ${outputChannels}u;
@compute @workgroup_size(${WORKGROUP_SIZE})
fn main(@builtin(workgroup_id) workgroupId: vec3u, @builtin(local_invocation_index) localIndex: u32) {
  let index = getInvocationIndex(workgroupId, localIndex);
  if (index >= INVOCATION_COUNT) { return; }
  let outputChannel = index / CELL_COUNT;
  let cell = index % CELL_COUNT;
  let row = i32(cell / SIZE);
  let column = i32(cell % SIZE);
  var sum = weightValues[OUTPUT_CHANNELS * INPUT_CHANNELS * 9u + outputChannel];
  for (var inputChannel = 0u; inputChannel < INPUT_CHANNELS; inputChannel++) {
    let weightBase = (outputChannel * INPUT_CHANNELS + inputChannel) * 9u;
    let inputBase = inputChannel * CELL_COUNT;
    for (var tap = 0u; tap < 9u; tap++) {
      let sourceRow = row + i32(tap / 3u) - 1;
      let sourceColumn = column + i32(tap % 3u) - 1;
      if (sourceRow < 0 || sourceColumn < 0 || sourceRow >= i32(SIZE) || sourceColumn >= i32(SIZE)) { continue; }
      sum += weightValues[weightBase + tap] * inputValues[inputBase + u32(sourceRow) * SIZE + u32(sourceColumn)];
    }
  }
  outputValues[index] = max(sum, 0.0);
}`,
    getReference: outputIndex => {
      const outputChannel = Math.floor(outputIndex / cellCount);
      const cell = outputIndex % cellCount;
      const row = Math.floor(cell / size);
      const column = cell % size;
      let sum = weightValues[outputChannels * inputChannels * 9 + outputChannel];
      for (let inputChannel = 0; inputChannel < inputChannels; inputChannel++) {
        for (let tap = 0; tap < 9; tap++) {
          const sourceRow = row + Math.floor(tap / 3) - 1;
          const sourceColumn = column + (tap % 3) - 1;
          if (sourceRow < 0 || sourceColumn < 0 || sourceRow >= size || sourceColumn >= size) {
            continue;
          }
          sum +=
            weightValues[(outputChannel * inputChannels + inputChannel) * 9 + tap] *
            inputValues[inputChannel * cellCount + sourceRow * size + sourceColumn];
        }
      }
      return Math.max(sum, 0);
    }
  };
  return probeCase;
}

async function runProbeCase(device: GPUDevice, probeCase: ProbeCase) {
  const workgroupCount = Math.ceil(probeCase.invocationCount / WORKGROUP_SIZE);
  const dispatchX = Math.min(workgroupCount, MAXIMUM_DISPATCH_X);
  const dispatchY = Math.ceil(workgroupCount / dispatchX);
  const createStorage = (values: Float32Array | null, length: number) => {
    const buffer = device.createBuffer({
      size: length * 4,
      usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST | GPUBufferUsage.COPY_SRC
    });
    if (values) device.queue.writeBuffer(buffer, 0, values);
    return buffer;
  };
  const inputBuffer = createStorage(probeCase.inputValues, probeCase.inputValues.length);
  const weightBuffer = createStorage(probeCase.weightValues, probeCase.weightValues.length);
  const outputBuffer = createStorage(null, probeCase.outputLength);
  const compileStarted = performance.now();
  const pipeline = await device.createComputePipelineAsync({
    layout: 'auto',
    compute: {
      module: device.createShaderModule({code: probeCase.getSource(dispatchX)}),
      entryPoint: 'main'
    }
  });
  const compileMilliseconds = performance.now() - compileStarted;
  const bindGroup = device.createBindGroup({
    layout: pipeline.getBindGroupLayout(0),
    entries: [
      {binding: 0, resource: {buffer: inputBuffer}},
      {binding: 1, resource: {buffer: weightBuffer}},
      {binding: 2, resource: {buffer: outputBuffer}}
    ]
  });
  const submit = async () => {
    const encoder = device.createCommandEncoder();
    const pass = encoder.beginComputePass();
    pass.setPipeline(pipeline);
    pass.setBindGroup(0, bindGroup);
    pass.dispatchWorkgroups(dispatchX, dispatchY);
    pass.end();
    const started = performance.now();
    device.queue.submit([encoder.finish()]);
    await device.queue.onSubmittedWorkDone();
    return performance.now() - started;
  };
  await device.queue.onSubmittedWorkDone();
  const firstMilliseconds = await submit();
  const samples: number[] = [];
  for (let iteration = 0; iteration < 5; iteration++) samples.push(await submit());
  samples.sort((first, second) => first - second);

  // A second clock that cannot be early: five dispatches and then a mapped read of the last output.
  const verifiedStarted = performance.now();
  for (let iteration = 0; iteration < 5; iteration++) {
    const encoder = device.createCommandEncoder();
    const pass = encoder.beginComputePass();
    pass.setPipeline(pipeline);
    pass.setBindGroup(0, bindGroup);
    pass.dispatchWorkgroups(dispatchX, dispatchY);
    pass.end();
    device.queue.submit([encoder.finish()]);
  }
  const verifiedReadBuffer = device.createBuffer({
    size: 4,
    usage: GPUBufferUsage.MAP_READ | GPUBufferUsage.COPY_DST
  });
  const verifiedEncoder = device.createCommandEncoder();
  verifiedEncoder.copyBufferToBuffer(
    outputBuffer,
    (probeCase.outputLength - 1) * 4,
    verifiedReadBuffer,
    0,
    4
  );
  device.queue.submit([verifiedEncoder.finish()]);
  await verifiedReadBuffer.mapAsync(GPUMapMode.READ);
  const readbackVerifiedMilliseconds = (performance.now() - verifiedStarted) / 5;
  verifiedReadBuffer.destroy();

  // Check a few outputs so that a kernel the compiler removed cannot report a time.
  const checkedIndices = [
    0,
    1,
    Math.floor(probeCase.outputLength / 3),
    Math.floor(probeCase.outputLength / 2) + 7,
    probeCase.outputLength - 1
  ];
  let maximumRelativeError = 0;
  for (const outputIndex of checkedIndices) {
    const readBuffer = device.createBuffer({
      size: 4,
      usage: GPUBufferUsage.MAP_READ | GPUBufferUsage.COPY_DST
    });
    const encoder = device.createCommandEncoder();
    encoder.copyBufferToBuffer(outputBuffer, outputIndex * 4, readBuffer, 0, 4);
    device.queue.submit([encoder.finish()]);
    await readBuffer.mapAsync(GPUMapMode.READ);
    const actual = new Float32Array(readBuffer.getMappedRange().slice(0))[0];
    readBuffer.destroy();
    const expected = probeCase.getReference(outputIndex);
    maximumRelativeError = Math.max(
      maximumRelativeError,
      Math.abs(actual - expected) / Math.max(1, Math.abs(expected))
    );
  }
  inputBuffer.destroy();
  weightBuffer.destroy();
  outputBuffer.destroy();
  const medianMilliseconds = samples[2];
  return {
    case: probeCase.label,
    multiplyAccumulatesInBillions: probeCase.multiplyAccumulateCount / 1e9,
    inputMegabytes: (probeCase.inputValues.length * 4) / 1048576,
    outputMegabytes: (probeCase.outputLength * 4) / 1048576,
    compileMilliseconds: Math.round(compileMilliseconds),
    firstMilliseconds: Math.round(firstMilliseconds),
    medianMilliseconds: Math.round(medianMilliseconds * 10) / 10,
    readbackVerifiedMilliseconds: Math.round(readbackVerifiedMilliseconds * 10) / 10,
    billionMultiplyAccumulatesPerSecond:
      Math.round((probeCase.multiplyAccumulateCount / 1e9 / (medianMilliseconds / 1000)) * 10) / 10,
    maximumRelativeError
  };
}

it('reports the throughput of plain WGSL neural layers', {timeout: 600000}, async () => {
  const lumaDevice = await getWebGPUTestDevice();
  requireTestDevice(lumaDevice);
  const device = (lumaDevice as unknown as {handle: GPUDevice}).handle;
  const cases: ProbeCase[] = [
    createDenseElementCase(262144, 64, 64),
    createDenseRowCase(262144, 64, 64),
    createDenseElementCase(1048576, 16, 16),
    createDenseRowCase(1048576, 16, 16),
    createCoordinateNetworkCase(1048576, 8, 64),
    createCoordinateNetworkCase(4194304, 4, 32),
    createConvolutionCase(256, 32, 32),
    createConvolutionCase(512, 32, 32),
    createConvolutionCase(256, 64, 64),
    createConvolutionCase(1024, 8, 8)
  ];
  const report: unknown[] = [
    {
      limits: {
        maxStorageBufferBindingSize: device.limits.maxStorageBufferBindingSize,
        maxBufferSize: device.limits.maxBufferSize,
        maxComputeWorkgroupsPerDimension: device.limits.maxComputeWorkgroupsPerDimension,
        maxStorageBuffersPerShaderStage: device.limits.maxStorageBuffersPerShaderStage
      },
      features: [...device.features].filter(feature =>
        ['shader-f16', 'subgroups', 'timestamp-query'].includes(feature)
      ),
      device: lumaDevice.info
    }
  ];
  for (const probeCase of cases) {
    const result = await runProbeCase(device, probeCase);
    expect(result.maximumRelativeError).toBeLessThan(1e-3);
    report.push(result);
  }
  // Fails on purpose: the headless runner does not print console output, a failure message it does.
  throw new Error(`NEURAL_LAYER_PROBE ${JSON.stringify(report)}`);
});
