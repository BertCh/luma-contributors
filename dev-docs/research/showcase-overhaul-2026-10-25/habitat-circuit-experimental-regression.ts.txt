import {expect, it} from 'vitest';
import {getWebGPUTestDevice} from '@luma.gl/test-utils';
import {addRasterResistanceConnectivityRecipe} from '../../../src/gpu-spatial-analysis/landscape-ecology/sparse-resistance';
import fixtureData from '../../oracles/fixtures/resistance-residual-drift.json';
import {makeSpatialFixture, readFloats, readIntegers} from '../../spatial-test-utils';
import {requireTestDevice} from '../../utils/require-test-device';

// The original single-phase 1024 solve stopped at its recursive residual tolerance,
// but independently recomputed current conservation was 2.24e-4, above 1e-4.
// The fixture reference is a Float64 CPU solve of exactly rounded Float32 conductances.
for (const maximumIterations of [128, 256, 512, 1024]) {
  it(`sparse resistance certifies or withholds within budget ${maximumIterations}`, async () => {
    const device = await getWebGPUTestDevice('core');
    requireTestDevice(device);
    const fixture = makeSpatialFixture(device);
    try {
      const resistanceByClass = new Map(
        Object.entries(fixtureData.strongResistanceByClass).map(([category, resistance]) => [
          Number(category),
          resistance
        ])
      );
      const resistanceValues = Float32Array.from(
        fixtureData.runs.flatMap(([category, count]) =>
          Array.from({length: count}, () => resistanceByClass.get(category) ?? Number.NaN)
        )
      );
      expect(resistanceValues.length).toBe(fixtureData.width * fixtureData.height);
      expect(Array.from(resistanceValues).every(Number.isFinite)).toBe(true);
      const resistance = fixture.view('float32', resistanceValues);
      const observedStatus = fixture.view('uint32', new Uint32Array(4));
      const observedDiagnostics = fixture.view('float32', new Float32Array(4));
      const result = addRasterResistanceConnectivityRecipe(fixture.graph, {
        id: 'residual-drift',
        resistance: resistance.view,
        width: fixtureData.width,
        height: fixtureData.height,
        cellSize: [fixtureData.cellSize, fixtureData.cellSize],
        connectivity: 4,
        source: fixtureData.source,
        sink: fixtureData.sink,
        maximumIterations,
        tolerance: 0.0001
      });
      fixture.graph.addObservationTap(result.status, {
        afterNode: 'residual-drift-solve-certificate',
        sink: fixture.graph.createDataView(observedStatus.view.buffer, {
          format: 'uint32x4',
          length: 1
        })
      });
      fixture.graph.addObservationTap(result.outputs.diagnostics, {
        afterNode: 'residual-drift-current-density',
        sink: fixture.graph.createDataView(observedDiagnostics.view.buffer, {
          format: 'float32x4',
          length: 1
        })
      });
      const observedPotentials = fixture.view('float32', new Float32Array(resistanceValues.length));
      const observedConductances = fixture.view(
        'float32',
        new Float32Array(result.raster.conductances.length)
      );
      fixture.graph.addObservationTap(result.outputs.potentials, {
        afterNode: 'residual-drift-current-density',
        sink: observedPotentials.view
      });
      fixture.graph.addObservationTap(result.raster.conductances, {
        afterNode: 'residual-drift-current-density',
        sink: observedConductances.view
      });
      await fixture.run();
      const status = await readIntegers(observedStatus.buffer);
      const diagnostics = await readFloats(observedDiagnostics.buffer);
      if (maximumIterations === 1024) {
        // Captured independently from accepted scalars, for a Float64 edge-current audit.
        console.log(
          'RESISTANCE_GPU_DIAGNOSTIC',
          JSON.stringify({
            maximumIterations,
            status,
            diagnostics,
            potentials: await readFloats(observedPotentials.buffer),
            conductances: await readFloats(observedConductances.buffer)
          })
        );
      }
      expect(status[1]).toBe(maximumIterations);
      expect(status[2]).toBe(1);
      expect(status[3]).toBe(0);
      expect(result.provenance.maximumIterations).toBe(maximumIterations);
      expect(result.provenance.solverSpmvStrategy).toBe(
        maximumIterations >= 512 ? 'scalar-row' : 'adaptive-default'
      );
      expect(result.provenance.solverPhaseIterations).toEqual(
        maximumIterations >= 512 ? [maximumIterations - 64, 64] : [maximumIterations]
      );
      expect(result.provenance.solverPhaseIterations.reduce((sum, budget) => sum + budget, 0)).toBe(
        maximumIterations
      );
      if (maximumIterations < 512) {
        expect(status[0]).toBe(0);
        expect(Number.isNaN(diagnostics[0])).toBe(true);
        expect(diagnostics[1]).toBeGreaterThan(0.0001);
      } else {
        expect(status[0], JSON.stringify({maximumIterations, status, diagnostics})).toBe(1);
        expect(diagnostics[1]).toBeLessThanOrEqual(0.0001);
        expect(
          Math.abs(diagnostics[0] - fixtureData.float64Reference.effectiveResistance)
        ).toBeLessThan(0.2);
        expect(Math.abs(diagnostics[2] - fixtureData.float64Reference.energy)).toBeLessThan(0.2);
      }
    } finally {
      fixture.destroy();
    }
  });
}
